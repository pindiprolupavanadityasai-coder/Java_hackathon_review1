 public class SolarSystemDetails {
    public static void main(String[] args) {
        int panelID = 101;                  // integer
        double energyGenerated = 12.5;      // decimal value
        int numberOfPanels = 20;            // integer
        char systemStatus = 'A';            // character (e.g., A = Active)

        System.out.println("Rooftop Solar System Details:");
        System.out.println("Panel ID: " + panelID);
        System.out.println("Energy Generated (kWh): " + energyGenerated);
        System.out.println("Number of Panels: " + numberOfPanels);
        System.out.println("System Status: " + systemStatus);
    }
}
